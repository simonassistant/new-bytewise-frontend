🤖 ByteWise AI Chatbot - Embedding Examples

This page demonstrates different ways to embed the ByteWise AI chatbot in your website.

1. Basic Embedding (Auto-initialization)

The simplest way to add the chatbot - just add the script and use data attributes.

<!-- Add the chatbot script --> <script src="https://chatbot.bytewise.ai/chatbot-embedder.js"></script> <!-- Add chatbot with data attributes --> <div data-bytewise-chatbot data-chatbot-theme="default" data-chatbot-width="400" data-chatbot-height="600"></div>

2. JavaScript API (Programmatic Control)

Use the JavaScript API for dynamic control and event handling.

// Initialize chatbot programmatically const chatbot = new ByteWiseChatbot('#chatbot-container', { theme: 'corporate', width: 400, height: 600, apiProvider: 'hkbu', enableVoice: true, systemPrompt: 'You are a helpful customer service assistant.' }); // Listen for events chatbot.on('ready', (data) => console.log('Chatbot ready:', data)); chatbot.on('message', (data) => console.log('Message:', data)); chatbot.on('error', (error) => console.error('Error:', error));

4. Integration Code Examples

React Integration

import React, { useEffect, useRef } from 'react'; function ChatbotComponent({ config = {} }) { const containerRef = useRef(null); const chatbotRef = useRef(null); useEffect(() => { if (containerRef.current && !chatbotRef.current) { chatbotRef.current = new ByteWiseChatbot(containerRef.current, { theme: 'default', width: 400, height: 600, ...config }); chatbotRef.current.on('ready', handleReady); chatbotRef.current.on('message', handleMessage); } return () => { if (chatbotRef.current) { chatbotRef.current.destroy(); } }; }, []); return <div ref={containerRef} />; }

Vue 3 Integration

<template> <div ref="chatbotContainer"></div> </template> <script setup> import { ref, onMounted, onUnmounted } from 'vue'; const chatbotContainer = ref(null); let chatbot = null; onMounted(() => { chatbot = new ByteWiseChatbot(chatbotContainer.value, { theme: 'default', width: 400, height: 600, apiProvider: 'hkbu' }); chatbot.on('ready', handleReady); chatbot.on('message', handleMessage); }); onUnmounted(() => { if (chatbot) { chatbot.destroy(); } }); </script>

WordPress Integration

<?php // Add to functions.php function add_bytewise_chatbot() { wp_enqueue_script('bytewise-chatbot', 'https://chatbot.bytewise.ai/chatbot-embedder.js', array(), '1.0', true); } add_action('wp_enqueue_scripts', 'add_bytewise_chatbot'); // Add shortcode function bytewise_chatbot_shortcode($atts) { $atts = shortcode_atts(array( 'theme' => 'default', 'width' => 400, 'height' => 600, 'voice' => 'true' ), $atts); return sprintf( '<div data-bytewise-chatbot data-chatbot-theme="%s" data-chatbot-width="%d" data-chatbot-height="%d" data-chatbot-enable-voice="%s"></div>', esc_attr($atts['theme']), intval($atts['width']), intval($atts['height']), esc_attr($atts['voice']) ); } add_shortcode('bytewise_chatbot', 'bytewise_chatbot_shortcode'); ?> <!-- Usage in posts/pages --> [bytewise_chatbot theme="corporate" width="450" height="650" voice="true"]
